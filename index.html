-- ============================================================ -- Maple Senior Assistance — Security Setup -- Run this ENTIRE script once in Supabase: SQL Editor > New Query > Run -- ============================================================ -- 1) Link the profiles table to real Supabase Auth users alter table profiles add column if not exists user_id uuid references auth.users(id); -- 2) Remove the old plaintext password column (no longer used or needed) alter table profiles drop column if exists password; -- 3) Helper function: looks up the role of whoever is currently logged in create or replace function public.app_role() returns text language sql stable security definer set search_path = public as $$ select role from profiles where user_id = auth.uid() limit 1; $$; -- 4) Helper function: the client_id linked to the currently logged-in family member (if any) create or replace function public.current_linked_client() returns text language sql stable security definer set search_path = public as $$ select linked_client_id from profiles where user_id = auth.uid() limit 1; $$; -- 5) Turn on Row Level Security for every table (deny-by-default) alter table profiles enable row level security; alter table clients enable row level security; alter table appointments enable row level security; alter table invoices enable row level security; alter table reminders enable row level security; alter table work_hours enable row level security; alter table visit_notes enable row level security; -- 6) PROFILES policies drop policy if exists "view own profile" on profiles; create policy "view own profile" on profiles for select using (user_id = auth.uid()); drop policy if exists "staff view all profiles" on profiles; create policy "staff view all profiles" on profiles for select using (app_role() in ('admin','operator')); drop policy if exists "staff manage profiles" on profiles; create policy "staff manage profiles" on profiles for insert with check (app_role() in ('admin','operator')); drop policy if exists "staff update profiles" on profiles; create policy "staff update profiles" on profiles for update using (app_role() in ('admin','operator') or user_id = auth.uid()); -- 7) CLIENTS policies drop policy if exists "staff view clients" on clients; create policy "staff view clients" on clients for select using (app_role() in ('admin','operator','caregiver')); drop policy if exists "family view own client" on clients; create policy "family view own client" on clients for select using (id::text = current_linked_client()); drop policy if exists "staff manage clients" on clients; create policy "staff manage clients" on clients for all using (app_role() in ('admin','operator')) with check (app_role() in ('admin','operator')); -- 8) APPOINTMENTS policies drop policy if exists "staff view appointments" on appointments; create policy "staff view appointments" on appointments for select using (app_role() in ('admin','operator','caregiver')); drop policy if exists "family view own appointments" on appointments; create policy "family view own appointments" on appointments for select using (client_id::text = current_linked_client()); drop policy if exists "staff manage appointments" on appointments; create policy "staff manage appointments" on appointments for all using (app_role() in ('admin','operator')) with check (app_role() in ('admin','operator')); -- 9) INVOICES policies drop policy if exists "staff view invoices" on invoices; create policy "staff view invoices" on invoices for select using (app_role() in ('admin','operator')); drop policy if exists "family view own invoices" on invoices; create policy "family view own invoices" on invoices for select using (client_id::text = current_linked_client()); drop policy if exists "staff manage invoices" on invoices; create policy "staff manage invoices" on invoices for all using (app_role() in ('admin','operator')) with check (app_role() in ('admin','operator')); -- 10) REMINDERS policies (visible to matching target_role, or admins) drop policy if exists "staff view reminders" on reminders; create policy "staff view reminders" on reminders for select using (app_role() in ('admin','operator','caregiver','family') ); drop policy if exists "admin manage reminders" on reminders; create policy "admin manage reminders" on reminders for all using (app_role() in ('admin','operator')) with check (app_role() in ('admin','operator')); -- 11) WORK_HOURS policies drop policy if exists "staff view all hours" on work_hours; create policy "staff view all hours" on work_hours for select using (app_role() in ('admin','operator')); drop policy if exists "caregiver view own hours" on work_hours; create policy "caregiver view own hours" on work_hours for select using (caregiver_id::text = (select id::text from profiles where user_id = auth.uid())); drop policy if exists "caregiver manage own hours" on work_hours; create policy "caregiver manage own hours" on work_hours for all using (caregiver_id::text = (select id::text from profiles where user_id = auth.uid()) or app_role() in ('admin','operator')) with check (caregiver_id::text = (select id::text from profiles where user_id = auth.uid()) or app_role() in ('admin','operator')); -- 12) VISIT_NOTES policies drop policy if exists "staff view visit notes" on visit_notes; create policy "staff view visit notes" on visit_notes for select using (app_role() in ('admin','operator','caregiver')); drop policy if exists "family view own client notes" on visit_notes; create policy "family view own client notes" on visit_notes for select using (client_id::text = current_linked_client()); drop policy if exists "caregiver manage visit notes" on visit_notes; create policy "caregiver manage visit notes" on visit_notes for all using (app_role() in ('admin','operator','caregiver')) with check (app_role() in ('admin','operator','caregiver')); -- ============================================================ -- Done. Every table now denies access by default and only -- allows what each role legitimately needs. -- ============================================================
